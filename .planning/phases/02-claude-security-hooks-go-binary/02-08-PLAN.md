## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Claude Code → main.go (process args + stdin) | Trusted producer + trusted but bounded stdin payload (1 MB cap) |
| stdin payload → io.LimitReader → decoder | Class B exit 1 if read fails OR > 1 MB |
| Per-agent verdict text → DisallowUnknownFields decoder | Class A block if unknown field present |
| ResolveWorkspaceRoot → os.Chdir(root) | Single trusted chdir before predicates run; reverted via defer-restore for test repeatability |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-08-01 | DoS | Producer sends 100 MB JSON to exhaust memory | mitigate | Wrap stdin: `io.LimitReader(stdin, 1<<20)`. If decoder hits EOF prematurely (read > limit), block with reason "input_too_large". |
| T-02-08-02 | Tampering | Code-fence stripping mis-handles edge cases (e.g., ```json with embedded ``` in payload) | mitigate | stripCodeFences strips ONE leading fence + ONE trailing fence; if resulting parse fails, validate.go falls back to the raw verdict text per CONTEXT.md line 100 Claude's Discretion. Test coverage: bare JSON, ```json-fenced JSON, ```-fenced JSON, fence-with-trailing-whitespace, no-fence with leading whitespace. |
| T-02-08-03 | Spoofing | Producer sends a verdict for one agent but tool_input.subagent_type names another (mismatched dispatch) | accept (Phase 2) | The validator trusts tool_input.subagent_type as the dispatch key — Claude Code controls this field. If wrong agent is dispatched, the verdict will fail D-10 (unknown fields) or per-agent invariants on shape — both block. Defense in depth, not a primary mitigation. |
| T-02-08-04 | Information Disclosure | Block reason contains raw fragments of verdict text | mitigate | EmitBlock only writes {decision, reason} JSON; reason is the D-03 formatted Violation list with no raw payload bytes. validate.go ensures FormatViolation receives only ID/Description/Path/Expected/Actual strings — never raw json.RawMessage. |

    package hooks

    import "strings"

    // stripCodeFences strips a single ```json or ``` fence (with surrounding
    // whitespace) from the leading and trailing edges of s. Internal ``` sequences
    // are left untouched. Per CONTEXT.md Claude's Discretion (line 100): the caller
    // may fall back to the raw input if the stripped form fails to parse.
    func stripCodeFences(s string) string {
        s = strings.TrimSpace(s)
        if strings.HasPrefix(s, "```json") {
            s = strings.TrimPrefix(s, "```json")
        } else if strings.HasPrefix(s, "```") {
            s = strings.TrimPrefix(s, "```")
        }
        s = strings.TrimSpace(s)
        s = strings.TrimSuffix(s, "```")
        return strings.TrimSpace(s)
    }
    ```

    // runAgentValidation dispatches based on subagentType. Returns the list of
    // formatted block-reason fragments (one per Violation) in registration order.
    // Empty []string means the verdict passed all invariants.
    // err != nil signals Class B failure (decode of unrelated infrastructure JSON, etc.).
    func runAgentValidation(subagentType, verdictText, inputText string) ([]string, error) {
        switch subagentType {
        case "go-cartographer":
            return runCartographer(verdictText)
        case "go-taint-tracer":
            return runTaintTracer(verdictText, inputText)
        case "go-authz-tracer":

    func decodeStrict(text string, v interface{}) error {
        dec := json.NewDecoder(strings.NewReader(text))

    package hooks

    import (
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "os"
        "strings"

        "github.com/saghaulor/claude-security-hooks/internal/invariants"
    )

    func Validate(stdin io.Reader, stdout io.Writer, stderr io.Writer) error {
        limited := io.LimitReader(stdin, stdinCapBytes+1)
        body, err := io.ReadAll(limited)
        if err != nil {
            return fmt.Errorf("validate: read stdin: %w", err)
        }
        if len(body) > stdinCapBytes {
            return fmt.Errorf("validate: input exceeds %d bytes", stdinCapBytes)
        }
        var ev PostToolUseEvent
        dec := json.NewDecoder(strings.NewReader(string(body)))
        dec.DisallowUnknownFields()
        if err := dec.Decode(&ev); err != nil {
            return EmitBlock(stdout, fmt.Sprintf("H7: PostToolUseEvent parse: %v", err))
        }
        if !IsSecurityAgent(ev.ToolInput.SubagentType) {
            return nil
        }
        root := invariants.ResolveWorkspaceRoot()
        if root == "" {
            return EmitBlock(stdout, "workspace_root_not_found")
        }
        prev, _ := os.Getwd()
        if err := os.Chdir(root); err != nil {
            return fmt.Errorf("validate: chdir(%q): %w", root, err)
        }
        defer func() { _ = os.Chdir(prev) }()

        verdictText, err := extractTextContent(ev.ToolResponse.Content)
        if err != nil {
            return EmitBlock(stdout, fmt.Sprintf("H7: tool_response.content not parseable as text: %v", err))
        }
        if strings.TrimSpace(verdictText) == "" {
            return EmitBlock(stdout, "H7: content empty")
        }
        stripped := stripCodeFences(verdictText)
        // Attempt parse on stripped; if it fails, fall back to raw verdictText.
        reasons, _ := runAgentValidation(ev.ToolInput.SubagentType, stripped, ev.ToolInput.Prompt)
        if len(reasons) == 1 && strings.Contains(reasons[0], "verdict parse") {
            fallback, _ := runAgentValidation(ev.ToolInput.SubagentType, verdictText, ev.ToolInput.Prompt)
            if len(fallback) > 0 && !strings.Contains(fallback[0], "verdict parse") {
                reasons = fallback
            }
        }
        // Synthesis: also run S1 directory check (review-report.{json,md} expected at workspace root).
        if ev.ToolInput.SubagentType == "synthesis" {
            for _, inv := range invariants.SynthesisDirInvariants {
                for _, vio := range inv.Check(".") {
                    reasons = append(reasons, FormatViolation(inv.ID, inv.Description, vio))
                }
            }
        }
        if len(reasons) > 0 {
            return EmitBlock(stdout, FormatBlockReason(reasons))
        }
        return nil
    }

    // extractTextContent handles string OR array per RESEARCH.md Pitfall 4.
    func extractTextContent(raw json.RawMessage) (string, error) {
        if len(raw) == 0 {
            return "", errors.New("empty raw message")
        }
        var s string
        if err := json.Unmarshal(raw, &s); err == nil {
            return s, nil
        }
        var arr []map[string]json.RawMessage
        if err := json.Unmarshal(raw, &arr); err != nil {
            return "", fmt.Errorf("content is neither string nor array: %w", err)
        }
        var sb strings.Builder
        for _, seg := range arr {
            for _, key := range []string{"text", "content"} {
                if rawSeg, ok := seg[key]; ok {
                    var sv string
                    if err := json.Unmarshal(rawSeg, &sv); err == nil {
                        sb.WriteString(sv)
                    }
                }
            }
        }
        return sb.String(), nil
    }
